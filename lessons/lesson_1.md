# Подключение скриптов на веб-страницу
1. Порядок загрузки: скрипт подключается в конце <body>, чтобы не блокировать отрисовку HTML.
2. Внешние файлы (.js) - предпочтительнее.
3. Переиспользование на многих страницах.
4. Лаконичность кода.

# Синтаксис
1. Точка с запятой ; - рекомендуется ставить
2. Блоки кода: {}
3. Комментарии: // - однострочный комментарий /* */ - многострочный комментарий
4. Интерполяция (форматированная строка) ``

# Переменные
1. Объявление let/const (var - устаревшее)
2. Имена переменных: 
create user avatar
snake_case - create_user_avatar 
camelCase - createUserAvatar
3. Регистрозависимость let A и let a - две разных переменных
4. Из уникального - могут содержать символы _, $
5. Не могут начинаться с цифры
6. Не могут совпадать с ключевыми словами

# Типы данных
1. Примитивы: String, Number, Boolean, undefined (значение не присвоено), null (пустое значение)
2. Объекты - коллекция свойств
3. Оператор typeof - проверить тип данных переменной

# Операторы
1. Арифметические: +, -, *, /, %, ** 
2. Сравнение: 
== - проверка равенства 
=== - проверка равенства с учетом типа данных
аналогично: !=, !==, >, <, >=, <=
3. Инкрементация: ++, --
4. Присваивание +=, -=

# Массивы - упорядоченный список данных
1. let fruits = ['apple', 'orange', 'banana', 32, false];
2. Обращение по индексу fruits[0];
3. Могут хранить элементы любых типов

# if...else / switch
let income = 100;
if (income > 100) {
    console.log("Доход больше 100");
}
else if (income === 100) {
    console.log("Доход равен 100")
}
else {
    console.log("Доход меньше 100")
}

# Циклы
let counter = 0;
while (counter < 10) {
    counter++;  // counter = counter + 1
    console.log(counter);
}

for (let i = 0; i < 10; i++) {
    console.log(i);
}

# Преобразование типов
## Явное преобразование 
let number = 42;
let string = "3.14";

let result1 = number + Number(string);
console.log(result1);

## Неявное преобразование - когда JS автоматически преобразует типы
console.log("5" + 3);
console.log("5" - 3);
console.log("5" * "2");
